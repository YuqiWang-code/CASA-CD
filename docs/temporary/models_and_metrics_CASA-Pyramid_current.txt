"""Measure full-batch training memory and throughput for CASA variants."""
import argparse
import gc
import json
import os
import subprocess
import time
from pathlib import Path

import torch

from model.casa_net import CASANet
from train_CASA import PROTOCOL, NO_DECAY_PROJECTIONS, make_loader, training_losses, make_optimizer


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--dataset-root", type=Path, required=True)
    parser.add_argument("--pretrained", required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    torch.set_num_threads(4)
    torch.backends.cudnn.benchmark = True
    torch.backends.cudnn.allow_tf32 = True
    torch.backends.cuda.matmul.allow_tf32 = False
    image, target = next(iter(make_loader(args.dataset_root, "train")))
    image, target = image.cuda().float(), target.cuda().float()
    results = {}
    for variant in ["none", "hyperbolic", "qkv"]:
        torch.manual_seed(2333)
        model = CASANet(args.pretrained, variant).cuda().train()
        optimizer = make_optimizer(model)
        torch.cuda.reset_peak_memory_stats()
        times = []
        for step in range(6):
            torch.cuda.synchronize()
            start = time.time()
            optimizer.zero_grad(set_to_none=True)
            logits, auxiliary = model(image[:, :3], image[:, 3:], return_aux=True)
            loss, main_loss, auxiliary_loss, edge_loss = training_losses(logits, auxiliary, target)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), float("inf"), error_if_nonfinite=True, foreach=True)
            optimizer.step()
            torch.cuda.synchronize()
            times.append(time.time() - start)
        model.context.collect_diag = True
        with torch.no_grad():
            model.predict(image[:, :3], image[:, 3:])
        used = subprocess.check_output(
            ["nvidia-smi", "--query-compute-apps=pid,used_memory", "--format=csv,noheader,nounits"],
            text=True)
        process_mib = next(float(row.split(",")[1]) for row in used.splitlines()
                           if int(row.split(",")[0]) == os.getpid())
        context_mib = process_mib - torch.cuda.memory_reserved() / 1024**2
        result = dict(protocol=PROTOCOL, batch_size=image.shape[0],
                      training_outputs=4, training_prediction_format="logits",
                      guide_weight=0.2, boundary_weight=0.05, encoder_lr_mult=0.1,
                      inference_method="predict", inference_outputs=1,
                      grid_sizes=list(model.context.grid_sizes),
                      contexts_per_scale=list(model.context.tokens_per_scale),
                      joint_contexts=sum(model.context.tokens_per_scale),
                      qk_geometry=model.context.geometry, value_geometry=model.context.value_geometry,
                      context_projection_weight_decay=1e-4, context_other_weight_decay=0.0,
                      no_decay_projections=[n for n, _ in model.named_parameters() if n in NO_DECAY_PROJECTIONS],
                      peak_allocated_mib=torch.cuda.max_memory_allocated() / 1024**2,
                      peak_reserved_mib=torch.cuda.max_memory_reserved() / 1024**2,
                      context_overhead_mib=context_mib, process_memory_mib=process_mib,
                      seconds_per_step=sum(times[2:]) / len(times[2:]), loss=float(loss.detach()),
                      params=sum(p.numel() for p in model.parameters()), diagnostics=model.diagnostics())
        results[variant] = result
        print(f"[PROFILE] {variant} {json.dumps(result)}", flush=True)
        del model, optimizer, logits, auxiliary, loss, main_loss, auxiliary_loss, edge_loss
        gc.collect()
        torch.cuda.empty_cache()
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(results, indent=2), encoding="utf-8")


if __name__ == "__main__":
    main()

"""Native pyramid, paired supervision, shared initialization and deployment."""
import torch

from model.casa_net import CASANet
from train_CASA import context_decay_names, training_losses, make_optimizer, verify_deployment


def main():
    torch.set_num_threads(4)
    torch.manual_seed(2333)
    a, b = torch.randn(1, 3, 256, 256), torch.randn(1, 3, 256, 256)
    target = torch.randint(0, 2, (1, 1, 256, 256)).float()
    reference = None
    counts = {"none": (2047190, 1935875), "hyperbolic": (2073620, 1962305),
              "qkv": (2085964, 1974649)}
    for variant in ("none", "hyperbolic", "qkv"):
        torch.manual_seed(2333)
        model = CASANet(module=variant).eval()
        count = sum(p.numel() for p in model.parameters())
        assert count == counts[variant][0], count
        with torch.no_grad():
            encoded = model.encoder(a)[1:]
            assert [tuple(x.shape[1:]) for x in encoded] == [(24,64,64),(32,32,32),(96,16,16),(320,8,8)]
            logits, auxiliary = model(a, b, return_aux=True)
            outputs = model(a, b)
            assert len(logits) == len(outputs) == len(auxiliary["guide_logits"]) == 4
            assert len(auxiliary["boundary_logits"]) == 2
            assert all(x.shape == target.shape for x in logits)
            assert all(torch.equal(z.sigmoid(), p) for z, p in zip(logits, outputs))
            assert torch.equal(model.predict(a, b), outputs[0])
            if reference is None:
                reference = outputs[0]
            else:
                assert torch.equal(reference, outputs[0]), "common path initialization differs"
        model.train()
        optimizer = make_optimizer(model)
        context_ids = {id(p) for p in model.context.parameters()}
        encoder_ids = {id(p) for p in model.encoder.parameters()}
        projection_ids = {id(p) for name, p in model.named_parameters() if name in context_decay_names(model)}
        grouped = [id(p) for group in optimizer.param_groups for p in group["params"]]
        assert len(grouped) == len(set(grouped)) == len(list(model.parameters()))
        for group in optimizer.param_groups:
            assert all(group["weight_decay"] == (0.0 if id(p) in context_ids and id(p) not in projection_ids
                                                 else 1e-4) for p in group["params"])
            assert all(group["lr_scale"] == (.1 if id(p) in encoder_ids else 1.) for p in group["params"])
        for _ in range(3):
            optimizer.zero_grad(set_to_none=True)
            logits, auxiliary = model(a, b, return_aux=True)
            loss, main_loss, guide, boundary = training_losses(logits, auxiliary, target)
            assert torch.allclose(loss, main_loss + .2 * guide + .05 * boundary)
            loss.backward()
            assert all(torch.isfinite(p.grad).all() for p in model.parameters() if p.grad is not None)
            optimizer.step()
        assert model.context.guide.weight.grad.abs().sum() > 0
        assert model.context.boundary.weight.grad.abs().sum() > 0
        if variant != "none":
            assert (model.context.gamma_context.abs() > 0).all()
            assert (model.context.gamma_detail.abs() > 0).all()
            assert all(layer.weight.grad.abs().sum() > 0 for layer in model.context.pair_proj)
        model.eval()
        _, report = verify_deployment(model, torch.cat((a, b), 1))
        assert max(report["per_scale_max_diff"]) < 2e-4
        assert report["binary_disagreement"] == 0
        deployed = model.switch_to_deploy()
        assert sum(p.numel() for p in deployed.parameters()) == counts[variant][1]
        print(variant, "parameters", count, "paired auxiliary gradients and deployment PASS")


if __name__ == "__main__":
    main()

"""Optimizer/RNG continuation and deployment backend-state checks."""
import copy
import random

import numpy as np
import torch
from torch import nn

from train_CASA import (PROTOCOL, make_optimizer, restore_training_state,
                        rng_state, strict_fold_precision, update_learning_rate,
                        learning_rate, context_decay_names, NO_DECAY_PROJECTIONS)


class SmallModel(nn.Module):
    def __init__(self):
        super().__init__()
        self.module = "qkv"
        self.encoder = nn.Linear(3, 3)
        self.body = nn.Linear(3, 3)
        self.context = nn.ModuleDict({"qk": nn.Linear(3, 3, bias=False),
                                      "k_contrast": nn.Linear(3, 3, bias=False),
                                      "v_contrast": nn.Linear(3, 3, bias=False),
                                      "v": nn.Linear(3, 3, bias=False),
                                      "out": nn.Linear(3, 3, bias=False),
                                      "guide": nn.Linear(3, 3)})

    def forward(self, x):
        x = self.encoder(x)
        return self.context["out"](self.context["qk"](self.body(x)) + self.context["v"](x)
                                   + self.context["k_contrast"](x) + self.context["v_contrast"](x)) + self.context["guide"](x)


def step(model, optimizer):
    x = torch.randn(4, 3) + random.random() + float(np.random.random())
    optimizer.zero_grad(set_to_none=True)
    model(x).square().mean().backward()
    optimizer.step()


def backend_state():
    return (torch.backends.cudnn.benchmark, torch.backends.cudnn.allow_tf32,
            torch.backends.cuda.matmul.allow_tf32, torch.get_float32_matmul_precision())


def main():
    torch.manual_seed(2333)
    np.random.seed(2333)
    random.seed(2333)
    model = SmallModel()
    optimizer = make_optimizer(model)
    decay = {id(p): group["weight_decay"] for group in optimizer.param_groups for p in group["params"]}
    scales = {id(p): group["lr_scale"] for group in optimizer.param_groups for p in group["params"]}
    decay_names = context_decay_names(model)
    for name, parameter in model.named_parameters():
        expected_decay = 0.0 if name.startswith("context.") and name not in decay_names else 1e-4
        assert decay[id(parameter)] == expected_decay
        assert scales[id(parameter)] == (0.1 if name.startswith("encoder.") else 1.0)
        if name in NO_DECAY_PROJECTIONS:
            assert decay[id(parameter)] == 0.0
        elif name.endswith('.weight'):
            assert decay[id(parameter)] == 1e-4
    for count in (0, 185, 199, 200, 20000, 39999):
        rate = learning_rate(5e-4, 1, count, 40000)
        update_learning_rate(optimizer, rate)
        assert optimizer.param_groups[0]["lr"] == rate * 0.1
        assert all(group["lr"] == rate for group in optimizer.param_groups[1:])
    for _ in range(4):
        step(model, optimizer)
    checkpoint = copy.deepcopy(dict(
        epoch=2, step=4, state_dict=model.state_dict(), optimizer=optimizer.state_dict(),
        manifest=dict(protocol=PROTOCOL, module="qkv", actual_steps=6, encoder_lr_mult=0.1), rng_state=rng_state()))
    for _ in range(2):
        step(model, optimizer)
    restored = SmallModel()
    restored_optimizer = make_optimizer(restored)
    epoch, count, exact_rng = restore_training_state(checkpoint, restored, restored_optimizer, 6, 2)
    assert (epoch, count, exact_rng) == (2, 4, True)
    for _ in range(2):
        step(restored, restored_optimizer)
    assert all(torch.equal(a, b) for a, b in zip(model.parameters(), restored.parameters()))
    assert all(int(state["step"]) == 6 for state in restored_optimizer.state.values())
    checkpoint.pop("rng_state")
    epoch, count, exact_rng = restore_training_state(checkpoint, restored, restored_optimizer, 6, 2)
    assert (epoch, count, exact_rng) == (2, 4, False)
    assert all(torch.equal(restored.state_dict()[key], value)
               for key, value in checkpoint["state_dict"].items())
    before = backend_state()
    try:
        with strict_fold_precision():
            assert backend_state() == (False, False, False, "highest")
            raise ValueError("exercise context exit")
    except ValueError:
        pass
    assert backend_state() == before
    print("Encoder LR ratio, context decay, optimizer/RNG continuation and backend restoration PASS")


if __name__ == "__main__":
    main()

"""RGB/date ordering, synchronized augmentation and stable binary loss checks."""
import tempfile
from pathlib import Path
from unittest.mock import patch

import cv2
import numpy as np

from torch.nn import functional as F

from dataset import casa_transforms as transforms
from train_CASA import (MEAN, STD, learning_rate, loss_function, make_loader,
                        mask_boundary, boundary_loss, training_losses, SpatialMetrics)


def main():
    with tempfile.TemporaryDirectory() as directory:
        root = Path(directory)
        for name in ("A", "B", "label", "list"):
            (root / name).mkdir()
        a = np.full((256, 256, 3), (17, 83, 211), dtype=np.uint8)
        b = np.full_like(a, (199, 67, 29))
        mask = np.zeros((256, 256), dtype=np.uint8)
        mask[100:180, 130:210] = 255
        cv2.imwrite(str(root / "A" / "pair.png"), cv2.cvtColor(a, cv2.COLOR_RGB2BGR))
        cv2.imwrite(str(root / "B" / "pair.png"), cv2.cvtColor(b, cv2.COLOR_RGB2BGR))
        cv2.imwrite(str(root / "label" / "pair.png"), mask)
        (root / "list" / "test.txt").write_text("pair.png\n")
        image, target = next(iter(make_loader(root, "test", batch_size=1, workers=0)))
        expected = (torch.tensor([17, 83, 211, 199, 67, 29]) / 255 - torch.tensor(MEAN)) / torch.tensor(STD)
        assert torch.allclose(image[0, :, 0, 0], expected, atol=1e-6)
        assert torch.equal(target[0, 0], torch.from_numpy((mask >= 128).astype(np.int64)))

    mask = np.zeros((32, 32), dtype=np.uint8)
    mask[5:19, 12:25] = 255
    pair = np.stack((mask, mask, mask, 255-mask, 255-mask, 255-mask), -1)
    with patch.object(transforms.random, "random", return_value=0.0), \
            patch.object(transforms.random, "randrange", return_value=1):
        image, target = transforms.Compose([
            transforms.RandomFlip(), transforms.RandomRotate90(), transforms.RandomExchange()
        ])(pair, mask)
    assert np.array_equal(image[:, :, 0], 255-target)
    assert np.array_equal(image[:, :, 3], target)
    with patch.object(transforms.random, "random", return_value=0.0), \
            patch.object(transforms.random, "randint", side_effect=[1, 2, 3, 4]):
        cropped, crop_target = transforms.RandomCropResize(8)(pair, mask)
    assert np.array_equal(crop_target, cv2.resize(mask[1:-2, 3:-4], (32, 32), interpolation=cv2.INTER_NEAREST))
    assert np.array_equal(cropped, cv2.resize(pair[1:-2, 3:-4], (32, 32)))

    logits = torch.tensor([[[[-100.0, 100.0]]]], requires_grad=True)
    target = torch.tensor([[[[1.0, 0.0]]]])
    loss = loss_function((logits,), target)
    loss.backward()
    assert torch.isfinite(loss) and torch.isfinite(logits.grad).all()
    assert (logits.grad.abs() > 0.4).all()
    moderate = torch.tensor([[[[-2.0, 1.0]]]])
    probability = moderate.sigmoid()
    expected = F.binary_cross_entropy(probability, target) + 1 - (
        2 * (probability * target).sum() + 1e-5) / (probability.sum() + target.sum() + 1e-5)
    assert torch.allclose(loss_function((moderate,), target), expected)
    assert learning_rate(5e-4, 1, 185, 40145) == 5e-4 * .9 * 186 / 200 + 5e-5

    for value in (0.0, 1.0):
        constant = torch.full((1, 1, 16, 16), value)
        assert mask_boundary(constant).count_nonzero() == 0
    target = torch.zeros(1, 1, 16, 16)
    target[:, :, 4:12, 5:13] = 1
    edge = mask_boundary(target)
    assert edge[0, 0, 3:13, 4:14].sum() == 64
    assert edge[:, :, 0].sum() == 0 and edge[:, :, -1].sum() == 0
    assert F.adaptive_max_pool2d(edge, (4, 4)).sum() > 0
    boundary_logits = (torch.zeros(1, 1, 4, 4, requires_grad=True),
                       torch.zeros(1, 1, 2, 2, requires_grad=True))
    aux = dict(guide_logits=tuple(torch.zeros(1, 1, side, side, requires_grad=True)
                                 for side in (8, 4, 2, 1)), boundary_logits=boundary_logits)
    outputs = tuple(torch.zeros_like(target, requires_grad=True) for _ in range(4))
    total, main_loss, guide, boundary = training_losses(outputs, aux, target)
    assert torch.equal(total, main_loss + .2 * guide + .05 * boundary)
    total.backward()
    assert all(z.grad is not None and torch.isfinite(z.grad).all() and z.grad.abs().sum() > 0
               for z in (*outputs, *aux["guide_logits"], *boundary_logits))
    extreme = tuple(torch.full_like(z, -100, requires_grad=True) for z in boundary_logits)
    edge_loss = boundary_loss(extreme, target)
    edge_loss.backward()
    assert torch.isfinite(edge_loss) and all(torch.isfinite(z.grad).all() for z in extreme)

    metric = SpatialMetrics(torch.device("cpu"))
    metric.update(target > 0, target)
    scores = metric.scores()
    assert scores["boundary_1px"]["F1"] == 1 and scores["boundary_2px"]["F1"] == 1
    assert scores["interior"]["recall"] == 1 and scores["interior"]["target_pixels"] == 16
    shifted = torch.roll(target, shifts=1, dims=-1)
    metric = SpatialMetrics(torch.device("cpu"))
    metric.update(shifted > 0, target)
    assert metric.scores()["boundary_1px"]["F1"] == 1
    metric = SpatialMetrics(torch.device("cpu"))
    metric.update(torch.zeros_like(target).bool(), target)
    assert metric.scores()["boundary_1px"]["F1"] == 0
    assert metric.scores()["interior"]["recall"] == 0
    small = torch.zeros_like(target)
    small[..., 8, 8] = 1
    metric = SpatialMetrics(torch.device("cpu"))
    metric.update(torch.zeros_like(small).bool(), small)
    assert metric.scores()["interior"]["target_pixels"] == 0
    assert metric.scores()["interior"]["recall"] is None
    print("RGB/date order, synchronized augmentation, stable losses, boundary targets/metrics and warmup PASS")

"""Paired Lorentz geometry: identities, reconstruction and FP32 coincidence gradients."""
import torch

from model.layers.lorentz_pair_ops import (
    euclidean_pairwise_distance, lorentz_inner, lorentz_logmap, lorentz_midpoint, lorentz_normalize,
    pair_midpoint_difference, pairwise_lorentz_distance, parallel_transport_to_origin,
)
from model.layers.poincare_ops import aligned_distance, expmap0, to_lorentz


def tangent_exp(base, tangent):
    squared = lorentz_inner(tangent, tangent, keepdim=True).clamp_min(0)
    norm = squared.clamp_min(1e-12).sqrt()
    return norm.cosh() * base + (norm.sinh() / norm) * tangent


def geometric_identities():
    torch.manual_seed(31)
    a = torch.randn(2, 4, 13, 8, dtype=torch.float64)
    b = torch.randn_like(a)
    a = a / a.norm(dim=-1, keepdim=True) * torch.linspace(0.01, 2.3, 13).double()[None, None, :, None]
    b = b / b.norm(dim=-1, keepdim=True) * 1.2
    x, y = to_lorentz(expmap0(a)), to_lorentz(expmap0(b))
    midpoint, difference = pair_midpoint_difference(x, y)
    reversed_midpoint, reversed_difference = pair_midpoint_difference(y, x)
    direct_log = lorentz_logmap(midpoint, y)
    direct_difference = parallel_transport_to_origin(midpoint, direct_log)
    assert torch.allclose(lorentz_inner(midpoint, midpoint), -torch.ones_like(midpoint[..., 0]), atol=2e-10)
    assert torch.allclose(midpoint, lorentz_normalize(x + y), atol=2e-10, rtol=2e-10)
    assert torch.equal(midpoint, reversed_midpoint)
    assert torch.equal(difference, -reversed_difference)
    assert torch.allclose(difference, direct_difference, atol=2e-10, rtol=2e-10)
    assert torch.allclose(lorentz_inner(midpoint, direct_log), torch.zeros_like(midpoint[..., 0]), atol=2e-10)
    assert torch.allclose(difference.norm(dim=-1), aligned_distance(expmap0(a), expmap0(b)) * 0.5,
                          atol=2e-10, rtol=2e-10)
    assert torch.allclose(tangent_exp(midpoint, direct_log), y, atol=2e-9, rtol=2e-10)
    assert torch.allclose(tangent_exp(midpoint, -direct_log), x, atol=2e-9, rtol=2e-10)
    expected = aligned_distance(expmap0(a).unsqueeze(-2), expmap0(b).unsqueeze(-3))
    assert torch.allclose(pairwise_lorentz_distance(x, y), expected, atol=2e-10, rtol=2e-10)
    weights = torch.rand(2, 4, 7, 13, dtype=torch.float64)
    weights = weights / weights.sum(-1, keepdim=True)
    centroid = lorentz_normalize(weights @ midpoint)
    assert torch.allclose(lorentz_inner(centroid, centroid), -torch.ones_like(centroid[..., 0]), atol=2e-10)
    print("midpoint, Log/PT, half-distance, reconstruction and centroid identities PASS")


def finite_float32_gradients():
    torch.manual_seed(39)
    for radius in (0.0, 0.5, 2.3):
        for perturbation in (0.0, 1e-6, 0.1):
            a = torch.randn(1, 2, 11, 8)
            a = (a / a.norm(dim=-1, keepdim=True) * radius).requires_grad_()
            b = (a.detach() + torch.randn_like(a) * perturbation).requires_grad_()
            x, y = to_lorentz(expmap0(a)), to_lorentz(expmap0(b))
            midpoint, difference = pair_midpoint_difference(x, y)
            general_log = lorentz_logmap(x, y)
            distance = pairwise_lorentz_distance(x, y)
            assert all(torch.isfinite(t).all() for t in (midpoint, difference, general_log, distance))
            if perturbation == 0:
                assert torch.equal(midpoint, x)
                assert torch.count_nonzero(difference) == 0
                assert torch.count_nonzero(general_log) == 0
                assert torch.count_nonzero(distance.diagonal(dim1=-2, dim2=-1)) == 0
            reference = aligned_distance(expmap0(a), expmap0(b)) * 0.5
            assert torch.allclose(difference.norm(dim=-1), reference, atol=1e-4, rtol=2e-3)
            loss = difference.square().mean() + general_log.square().mean() + midpoint.mean() + distance.mean()
            loss.backward()
            assert torch.isfinite(a.grad).all() and torch.isfinite(b.grad).all()
    print("FP32 origin, identical pairs, near-identical pairs and clipping-radius gradients PASS")


def pair_derivative_check():
    torch.manual_seed(43)

def contrast_semantics():
    op = PairedEuclideanContext().double()
    feature = torch.randn(1, 64, 8, 8, dtype=torch.double, requires_grad=True)
    for level in (0., .01, .5, .99, 1.):
        probability = torch.full((1, 1, 8, 8), level, dtype=torch.double)
        mean, contrast = op.prototypes(feature, probability, 1)
        assert torch.allclose(mean, feature.mean((-2, -1)).unsqueeze(1), atol=1e-12)
        assert contrast.abs().max() < 1e-12, "constant guide must not create contrast content"
        value = project_tokens(op.v_contrast, contrast, 4)
        assert value.abs().max() < 1e-12
    probability = torch.zeros(1, 1, 8, 8, dtype=torch.double)
    probability[..., 0, 0] = 1
    mean, contrast = op.prototypes(feature, probability, 1)
    positive = (feature[..., 0, 0].unsqueeze(1) + mean) / 2
    negative = (feature.flatten(2).sum(-1).unsqueeze(1) - feature[..., 0, 0].unsqueeze(1) + mean) / 64
    assert torch.allclose(contrast, positive - negative, atol=1e-12)
    logits = torch.randn(1, 1, 8, 8, dtype=torch.double, requires_grad=True)
    _, contrast = op.prototypes(feature, logits.sigmoid(), 1)
    grads = torch.autograd.grad(contrast.square().sum(), (feature, logits))
    assert all(torch.isfinite(g).all() and g.abs().sum() > 0 for g in grads)
    assert binomial_highpass(torch.ones(1, 32, 8, 8)).abs().max() == 0
    print("mean/contrast shrinkage, zero constant-guide values and assignment gradients PASS")


def aligned_hyperbolic_tokens():
    op = PairedHyperbolicContext().double()
    midpoint = to_lorentz(expmap0(torch.randn(1, 4, 8, 8, 8, dtype=torch.double) * .1))
    unsigned = torch.rand(1, 4, 8, 8, 8, dtype=torch.double)
    feature = torch.randn(1, 64, 8, 8, dtype=torch.double)
    probability = torch.full((1, 1, 8, 8), .3, dtype=torch.double)
    weights = op.assignments(probability, 2)
    km, ku, value = op.compress(midpoint, unsigned, feature, weights, 2)
    assert torch.allclose(km[:, :, 0::2], km[:, :, 1::2], atol=1e-12)
    assert torch.allclose(ku[:, :, 0::2], ku[:, :, 1::2], atol=1e-12)
    assert torch.allclose(value[:, :, 0::2], value[:, :, 1::2], atol=1e-12)
    probability.zero_()
    probability[..., 0, 0] = 1
    weights = op.assignments(probability, 2)
    km, ku, value = op.compress(midpoint, unsigned, feature, weights, 2)
    wm = weights[:, 0, 1, :, None]
    expected_m = lorentz_normalize((regions(midpoint, 2)[:, :, 0] * wm[:, None]).sum(-2))
    expected_u = (regions(unsigned, 2)[:, :, 0] * wm[:, None]).sum(-2)
    expected_x = (feature_regions(feature, 2)[:, 0] * wm).sum(-2).unsqueeze(1)
    expected_v = project_tokens(op.v, expected_x, 4).squeeze(2)
    assert torch.allclose(km[:, :, 1], expected_m, atol=1e-12)
    assert torch.allclose(ku[:, :, 1], expected_u, atol=1e-12)
    assert torch.allclose(value[:, :, 1], expected_v, atol=1e-12)
    assert weights[0, 0, 1, 0] > .9999, "sparse change weight was diluted"
    print("uniform/change duplicate tokens and aligned midpoint/difference/value pooling PASS")


def scale_priors():
    e = PairedEuclideanContext()
    q, k = torch.zeros(1, 4, 3, 16), torch.zeros(1, 4, 112, 16)
    for bank in range(2):
        attention = e.attention(q, k, 0, bank)
        masses = torch.stack([part.sum(-1) for part in attention.split(e.regions_per_scale, -1)], -1)
        assert torch.allclose(masses, torch.full_like(masses, .25), atol=1e-6)
        assert torch.allclose(attention.sum(-1), torch.ones_like(attention[..., 0]))
    h = PairedHyperbolicContext()
    origin_q, origin_k = torch.zeros(1, 4, 3, 9), torch.zeros(1, 4, 224, 9)
    origin_q[..., 0] = origin_k[..., 0] = 1
    attention, _, _ = h.attention(origin_q, torch.zeros(1, 4, 3, 8), origin_k,
                                   torch.zeros(1, 4, 224, 8), 0)
    masses = torch.stack([part.sum(-1) for part in attention.split(h.tokens_per_scale, -1)], -1)
    assert torch.allclose(masses, torch.full_like(masses, .25), atol=1e-6)
    print("two independent 112-token softmaxes and 224-token source priors PASS")


def contexts():
    sizes, channels = (32, 16, 8, 4), (24, 32, 96, 320)
    for context_type, expected_count in ((PairedEuclideanContext, 39032), (PairedHyperbolicContext, 26688)):
        op = context_type()
        assert sum(p.numel() for p in op.parameters()) == expected_count
        a = [torch.randn(1, c, n, n, requires_grad=True) for c, n in zip(channels, sizes)]
        b = [torch.randn_like(x, requires_grad=True) for x in a]
        tar = [torch.randn(1, 64, n, n, requires_grad=True) for n in sizes]

        assert len(auxiliary["guide_logits"]) == 4 and len(auxiliary["boundary_logits"]) == 2
        op.gamma_context.data.fill_(.1)
        op.gamma_detail.data.fill_(.1)
        op.collect_diag = True
        output, _ = op(a, b, tar)
        exchanged, _ = op(b, a, tar)
        assert all(torch.allclose(x, y, atol=2e-6, rtol=1e-5) for x, y in zip(output, exchanged))
        # With TAR held fixed, paired geometry is unordered; TAR itself can remain ordered.
        if isinstance(op, PairedHyperbolicContext):
            pair = op.paired_features(a, b, tar)

            identical = op.paired_features(a, a, tar)
            assert all(x.abs().max() == 0 for x in identical[1])
        else:
            identical = op.paired_features(a, a, tar)
            assert all(x.abs().max() == 0 for x in identical[0])
        gradients = torch.autograd.grad(output[0].square().mean(), tar + a + b, retain_graph=True)
        assert all(torch.isfinite(g).all() and g.abs().sum() > 0 for g in gradients)
        sum(x.square().mean() for x in output).backward()
        for name, parameter in op.named_parameters():
            if parameter.ndim >= 2:
                assert parameter.grad is not None and torch.isfinite(parameter.grad).all(), name
                assert parameter.grad.abs().sum() > 0, name
        for index, original in enumerate(tar):
            diag = op.last_diag[f"stage{index + 1}"]
            assert diag["finite"] and diag["contexts"] == 224
            assert diag["context_residual_to_feature_rms"] > 0
            if index < 2:
                assert diag["local_residual_to_feature_rms"] > 0
        print(context_type.__name__, expected_count, "identity/swap/dense detail/cross-scale task gradients PASS")

def loss_function(outputs, target):
    loss = target.new_zeros(())
    for logits in outputs:
        output = logits.sigmoid()
        intersection = (output * target).sum()
        dice = (2 * intersection + 1e-5) / (output.sum() + target.sum() + 1e-5)
        loss = loss + F.binary_cross_entropy_with_logits(logits, target) + 1 - dice
    return loss


def guide_loss(logits, target):
    return sum(F.binary_cross_entropy_with_logits(z, F.adaptive_avg_pool2d(target, z.shape[-2:]))
               for z in logits)


def mask_boundary(target, radius=1):
    """Two-sided morphological boundary without an artificial image-frame edge."""
    padded = F.pad(target, (radius,) * 4, mode="replicate")
    kernel = 2 * radius + 1
    return F.max_pool2d(padded, kernel, 1) + F.max_pool2d(-padded, kernel, 1)


def boundary_loss(logits, target):
    edge = mask_boundary(target)
    return sum(loss_function((z,), F.adaptive_max_pool2d(edge, z.shape[-2:]))
               for z in logits)


def training_losses(outputs, aux, target):
    main = loss_function(outputs, target)
    guide = guide_loss(aux["guide_logits"], target)
    boundary = boundary_loss(aux["boundary_logits"], target)
    return main + GUIDE_WEIGHT * guide + BOUNDARY_WEIGHT * boundary, main, guide, boundary


class SpatialMetrics:
    """Dataset-level boundary matching and segmentation outside a fixed GT band."""
    def __init__(self, device):
        self.boundary_counts = torch.zeros((2, 4), dtype=torch.int64, device=device)
        self.interior_counts = torch.zeros(4, dtype=torch.int64, device=device)

    @torch.no_grad()
    def update(self, prediction, target):
        prediction, truth = prediction.bool(), target.bool()
        predicted_edge = mask_boundary(prediction.float()).bool()
        true_edge = mask_boundary(target.float()).bool()
        for index, radius in enumerate((1, 2)):
            kernel = 2 * radius + 1
            near_truth = F.max_pool2d(true_edge.float(), kernel, 1, radius).bool()
            near_prediction = F.max_pool2d(predicted_edge.float(), kernel, 1, radius).bool()
            self.boundary_counts[index] += torch.stack((
                (predicted_edge & near_truth).sum(), predicted_edge.sum(),
                (true_edge & near_prediction).sum(), true_edge.sum()))
        interior = mask_boundary(target.float(), radius=2) == 0
        self.interior_counts += torch.stack((
            (prediction & truth & interior).sum(), (prediction & ~truth & interior).sum(),
            (~prediction & truth & interior).sum(), (~prediction & ~truth & interior).sum()))

    def scores(self):
        def rates(matched_prediction, predicted, matched_truth, positives):
            precision = matched_prediction / predicted if predicted else None
            recall = matched_truth / positives if positives else None
            f1 = (2 * precision * recall / (precision + recall)
                  if precision is not None and recall is not None and precision + recall else
                  0.0 if positives else None)
            return dict(precision=precision, recall=recall, F1=f1,
                        predicted_pixels=predicted, target_pixels=positives)
        result = {f"boundary_{index+1}px": rates(*counts)
                  for index, counts in enumerate(self.boundary_counts.cpu().tolist())}
        tp, fp, fn, tn = self.interior_counts.cpu().tolist()
        result["interior"] = dict(**rates(tp, tp+fp, tp, tp+fn),
                                  tp=tp, fp=fp, fn=fn, tn=tn)
        result["definition"] = dict(boundary="3x3 morphological gradient, replicate padding",
                                    boundary_tolerance="Chebyshev distance in input pixels",
                                    interior="outside GT dilation2-minus-erosion2 band")
        return result


def make_loader(root, split, batch_size=32, workers=1):
    operations = [transforms.Scale(256, 256)]
    if split == "train":
        operations += [transforms.RandomCropResize(8), transforms.RandomFlip(), transforms.RandomRotate90(),
                       transforms.RandomExchange()]
    operations += [transforms.Normalize(MEAN, STD), transforms.ToTensor()]
    data = Dataset(file_root=str(root), list_path=str(root / "list" / f"{split}.txt"),
                   transform=transforms.Compose(operations))
    return torch.utils.data.DataLoader(data, batch_size=batch_size, shuffle=split == "train",
                                       num_workers=workers, pin_memory=False,
                                       drop_last=split == "train")


@torch.no_grad()
def evaluate(model, loader, device, return_diagnostics=False):
    model.eval()
    meter = ConfuseMatrixMeter(n_class=2)
    losses = []
    spatial = SpatialMetrics(device) if return_diagnostics else None
    for image, target in loader:
        image, target = image.to(device).float(), target.to(device).float()
        outputs, _ = model(image[:, :3], image[:, 3:], return_aux=True)
        model.context.collect_diag = False
        if not all(torch.isfinite(x).all() for x in outputs):
            raise FloatingPointError("non-finite evaluation prediction")
        losses.append(float(loss_function(outputs, target)))
        meter.update_cm(pr=(outputs[0] > 0).long().cpu().numpy(), gt=target.cpu().numpy())
        if spatial is not None:
            spatial.update(outputs[0] > 0, target)
    result = (sum(losses) / len(losses), meter.get_scores())
    return (*result, spatial.scores()) if spatial is not None else result


def learning_rate(base, epoch, step, total):
    rate = base * (1 - step / total) ** 0.9
    if step < 200:
        rate = base * 0.9 * (step + 1) / 200 + 0.1 * base
    return rate


def context_decay_names(model):
    return {f"{name}.weight" for name, layer in model.named_modules()
            if name.startswith("context.") and isinstance(layer, (torch.nn.Conv2d, torch.nn.Linear))
            and f"{name}.weight" not in NO_DECAY_PROJECTIONS}


def make_optimizer(model, encoder_lr_mult=0.1):
    encoder, decay, no_decay = [], [], []
    decay_names = context_decay_names(model)
    for name, parameter in model.named_parameters():
        group = (encoder if name.startswith("encoder.") else
                 no_decay if name.startswith("context.") and name not in decay_names else decay)
        group.append(parameter)
    groups = [dict(params=encoder, weight_decay=1e-4, lr_scale=encoder_lr_mult,
                   lr=5e-4*encoder_lr_mult, group_name="encoder"),
              dict(params=decay, weight_decay=1e-4, lr_scale=1.0, group_name="body_decay"),
              dict(params=no_decay, weight_decay=0.0, lr_scale=1.0, group_name="context_no_decay")]
    return torch.optim.Adam(groups, lr=5e-4, betas=(0.9, 0.99), eps=1e-8)


def update_learning_rate(optimizer, scheduled_base_lr):
    for group in optimizer.param_groups:
        group["lr"] = scheduled_base_lr * group["lr_scale"]


def rng_state():
    return dict(python=random.getstate(), numpy=np.random.get_state(),
                torch=torch.get_rng_state(),
                cuda=torch.cuda.get_rng_state_all() if torch.cuda.is_available() else None)


def restore_training_state(checkpoint, model, optimizer, actual_steps, train_batches):
    manifest = checkpoint["manifest"]
    epoch, step = checkpoint["epoch"], checkpoint["step"]
    if (manifest["protocol"] != PROTOCOL or manifest["module"] != model.module or
            manifest["actual_steps"] != actual_steps or step != epoch * train_batches or
            manifest["encoder_lr_mult"] != optimizer.param_groups[0]["lr_scale"] or
            not 0 <= step <= actual_steps):
        raise RuntimeError("checkpoint does not match the requested training budget/module")
    model.load_state_dict(checkpoint["state_dict"])
    optimizer.load_state_dict(checkpoint["optimizer"])
    state = checkpoint.get("rng_state")
    if state is not None:
        random.setstate(state["python"])
        np.random.set_state(state["numpy"])
        torch.set_rng_state(state["torch"])
        if state["cuda"] is not None:
            torch.cuda.set_rng_state_all(state["cuda"])
    else:
        random.seed(2333)
        np.random.seed(2333)
        torch.manual_seed(2333)
        torch.cuda.manual_seed_all(2333)
    return epoch, step, state is not None


@contextmanager
def strict_fold_precision():
    benchmark = torch.backends.cudnn.benchmark
    cudnn_tf32 = torch.backends.cudnn.allow_tf32
    matmul_tf32 = torch.backends.cuda.matmul.allow_tf32
    precision = torch.get_float32_matmul_precision()
    try:
        torch.backends.cudnn.benchmark = False
        torch.backends.cudnn.allow_tf32 = False
        torch.backends.cuda.matmul.allow_tf32 = False
        torch.set_float32_matmul_precision("highest")
        yield
    finally:
        torch.backends.cuda.matmul.allow_tf32 = matmul_tf32
        torch.set_float32_matmul_precision(precision)
        torch.backends.cudnn.allow_tf32 = cudnn_tf32
        torch.backends.cudnn.benchmark = benchmark


@torch.no_grad()
def verify_deployment(model, image):
    model.eval()
    deploy = copy.deepcopy(model).switch_to_deploy().eval()
    with strict_fold_precision():
        original = model(image[:, :3], image[:, 3:])
        folded = deploy(image[:, :3], image[:, 3:])
        if not all(torch.isfinite(x).all() for x in (*original, *folded)):
            raise FloatingPointError("non-finite deployment verification output")
        differences = [float((a-b).abs().max()) for a, b in zip(original, folded)]
        disagreement = int(((original[0] > 0.5) != (folded[0] > 0.5)).sum())
    report = dict(policy=DEPLOY_POLICY, tolerance=2e-4, max_diff=differences[0],
                  per_scale_max_diff=differences, binary_disagreement=disagreement)
    print(f"[DEPLOY-VERIFICATION] {json.dumps(report)}", flush=True)
    if differences[0] > 2e-4:
        raise RuntimeError(f"deploy folding mismatch: {differences[0]}")
    return deploy, report

import numpy as np
import torch
import random
import cv2
import numpy


class Scale(object):
    """Resize image and label to a fixed size"""

    def __init__(self, wi, he):
        self.w = wi
        self.h = he

    def __call__(self, img, label):
        img = cv2.resize(img, (self.w, self.h))
        label = cv2.resize(label, (self.w, self.h), interpolation=cv2.INTER_NEAREST)
        return [img, label]


class RandomCropResize(object):
    """Random crop followed by resize back to original size"""

    def __init__(self, crop_area):
        self.cw = crop_area
        self.ch = crop_area

    def __call__(self, img, label):
        if random.random() < 0.5:
            h, w = img.shape[:2]
            top, bottom = random.randint(0, self.ch), random.randint(0, self.ch)
            left, right = random.randint(0, self.cw), random.randint(0, self.cw)

            img_crop = img[top:h - bottom, left:w - right]
            label_crop = label[top:h - bottom, left:w - right]

            img_crop = cv2.resize(img_crop, (w, h))
            label_crop = cv2.resize(label_crop, (w, h), interpolation=cv2.INTER_NEAREST)
            return img_crop, label_crop

        return [img, label]


class RandomFlip(object):
    """Random horizontal and vertical flip"""

    def __call__(self, image, label):
        if random.random() < 0.5:
            image = cv2.flip(image, 0)
            label = cv2.flip(label, 0)
        if random.random() < 0.5:
            image = cv2.flip(image, 1)
            label = cv2.flip(label, 1)
        return [image, label]


class RandomExchange(object):
    """Randomly swap pre- and post-event images (for bi-temporal input)"""

    def __call__(self, image, label):
        if random.random() < 0.5:
            pre_img = image[:, :, 0:3]
            post_img = image[:, :, 3:6]
            image = numpy.concatenate((post_img, pre_img), axis=2)
        return [image, label]


class RandomRotate90(object):
    """Apply one shared quarter-turn rotation to both dates and the mask."""

    def __call__(self, image, label):
        k = random.randrange(4)
        return [np.rot90(image, k).copy(), np.rot90(label, k).copy()]


class Normalize(object):
    """Normalize both RGB dates and binarize the 0/255 change mask."""

    def __init__(self, mean, std):
        self.mean = mean
        self.std = std

    def __call__(self, image, label):
        image = image.astype(np.float32) / 255.0
        label = (label >= 128).astype(np.float32)

        for i in range(6):
            image[:, :, i] = (image[:, :, i] - self.mean[i]) / self.std[i]

        return [image, label]


class ToTensor(object):
    """Convert numpy arrays to PyTorch tensors"""

    def __init__(self, scale=1):
        self.scale = scale

    def __call__(self, image, label):
        if self.scale != 1:
            h, w = label.shape[:2]
            image = cv2.resize(image, (w, h))
            label = cv2.resize(
                label,
                (int(w / self.scale), int(h / self.scale)),
                interpolation=cv2.INTER_NEAREST
            )

        image = np.ascontiguousarray(image.transpose((2, 0, 1)))

        image_tensor = torch.from_numpy(image)
        label_tensor = torch.LongTensor(
            np.array(label, dtype=np.int_)
        ).unsqueeze(0)

        return [image_tensor, label_tensor]


class Compose(object):
    """Compose multiple transforms"""

    def __init__(self, transforms):
        self.transforms = transforms

    def __call__(self, *args):
        for t in self.transforms:
            args = t(*args)
        return args

import cv2
from os.path import join as osp
import numpy
import torch.utils.data


def read_list(list_path):
    """Read a list/*.txt file: one image filename per line.

    The server-side dataset layout is `<root>/{A,B,label}/` + `<root>/list/*.txt`.
    """
    with open(list_path, "r") as f:
        names = [line.strip() for line in f if line.strip()]
    return names


class Dataset(torch.utils.data.Dataset):
    """Bi-temporal change detection dataset in A/B/label + list format.

    Args:
        file_root: dataset root containing A/ B/ label/ dirs
        list_path: path to list/*.txt with image filenames
        transform: Compose of Transforms
    """
    def __init__(self, file_root, list_path, transform=None):
        self.file_list = read_list(list_path)

        self.pre_images = [osp(file_root, 'A', x) for x in self.file_list]
        self.post_images = [osp(file_root, 'B', x) for x in self.file_list]
        self.gts = [osp(file_root, 'label', x) for x in self.file_list]

        self.transform = transform

    def __len__(self):
        return len(self.pre_images)

    def __getitem__(self, idx):
        pre_image_name = self.pre_images[idx]
        label_name = self.gts[idx]
        post_image_name = self.post_images[idx]

        pre_image = cv2.cvtColor(cv2.imread(pre_image_name), cv2.COLOR_BGR2RGB)
        label = cv2.imread(label_name, 0)
        post_image = cv2.cvtColor(cv2.imread(post_image_name), cv2.COLOR_BGR2RGB)

        img = numpy.concatenate((pre_image, post_image), axis=2)

        if self.transform:
            [img, label] = self.transform(img, label)

        return img, label

"""Native MobileNetV2 features, TAR, joint pyramid context and DCR."""
import torch
from torch import nn
from torch.nn import functional as F

from model.mobilenet_v2 import mobilenet_v2
from model.str_tar import MultiScaleTAR
from model.str_dcr import DCRDecoder
from model.layers.pyramid_context import AuxiliaryContext
from model.layers.paired_euclidean_context import PairedEuclideanContext
from model.layers.paired_hyperbolic_context import PairedHyperbolicContext


class CASANet(nn.Module):
    def __init__(self, pretrained_path=None, module="none"):
        super().__init__()
        context_type = {"none": AuxiliaryContext, "hyperbolic": PairedHyperbolicContext,
                        "qkv": PairedEuclideanContext}[module]
        self.module = module
        self.encoder = mobilenet_v2(pretrained_path)
        self.tar = MultiScaleTAR((24, 32, 96, 320), dim=64,
                                 use_temporal_aux=True, use_dcr_aux=True, use_residual=True)
        self.decoder = DCRDecoder(dim=64, use_aux=True, use_residual=True)
        self.head = nn.Conv2d(64, 1, 1)
        with torch.random.fork_rng():
            torch.manual_seed(2333)
            self.context = context_type()

    def _decode(self, a, b):
        features_a, features_b = self.encoder(a)[1:], self.encoder(b)[1:]
        enhanced, auxiliary = self.context(features_a, features_b, self.tar(features_a, features_b))
        return self.decoder.forward_pyramid(enhanced), auxiliary

    def forward(self, a, b, return_aux=False):
        """Return four probabilities, or raw logits with guide and boundary heads."""
        pyramid, auxiliary = self._decode(a, b)
        logits = tuple(F.interpolate(self.head(x), size=a.shape[-2:],
                                     mode="bilinear", align_corners=False)
                       for x in pyramid)
        return (logits, auxiliary) if return_aux else tuple(x.sigmoid() for x in logits)

    def predict(self, a, b):
        """Return the final change probability without auxiliary prediction heads."""
        pyramid, _ = self._decode(a, b)
        logits = F.interpolate(self.head(pyramid[0]), size=a.shape[-2:],
                               mode="bilinear", align_corners=False)
        return logits.sigmoid()

    def switch_to_deploy(self):
        self.tar.switch_to_deploy()
        self.decoder.switch_to_deploy()
        return self

    def diagnostics(self):
        return dict(self.context.last_diag)

import numpy as np


###################       metrics      ###################
class AverageMeter(object):
    """Computes and stores the average and current value"""

    def __init__(self):
        self.initialized = False
        self.val = None
        self.avg = None
        self.sum = None
        self.count = None

    def initialize(self, val, weight):
        self.val = val
        self.avg = val
        self.sum = val * weight
        self.count = weight
        self.initialized = True

    def update(self, val, weight=1):
        if not self.initialized:
            self.initialize(val, weight)
        else:
            self.add(val, weight)

    def add(self, val, weight):
        self.val = val
        self.sum += val * weight
        self.count += weight
        self.avg = self.sum / self.count

    def value(self):
        return self.val

    def average(self):
        return self.avg

    def get_scores(self):
        scores_dict = cm2score(self.sum)
        return scores_dict

    def clear(self):
        self.initialized = False


###################      cm metrics      ###################
class ConfuseMatrixMeter(AverageMeter):
    """Computes and stores the average and current value"""

    def __init__(self, n_class):
        super(ConfuseMatrixMeter, self).__init__()
        self.n_class = n_class

    def update_cm(self, pr, gt, weight=1):
        """获得当前混淆矩阵，并计算当前F1得分，并更新混淆矩阵"""
        val = get_confuse_matrix(num_classes=self.n_class, label_gts=gt, label_preds=pr)
        self.update(val, weight)
        current_score = cm2F1(val)
        return current_score

    def get_scores(self):
        scores_dict = cm2score(self.sum)
        return scores_dict


def harmonic_mean(xs):
    harmonic_mean = len(xs) / sum((x + 1e-6) ** -1 for x in xs)
    return harmonic_mean


def cm2F1(confusion_matrix):
    hist = confusion_matrix
    tp = hist[1, 1]
    fn = hist[1, 0]
    fp = hist[0, 1]
    tn = hist[0, 0]
    # recall
    recall = tp / (tp + fn + np.finfo(np.float32).eps)
    # precision
    precision = tp / (tp + fp + np.finfo(np.float32).eps)
    # F1 score
    f1 = 2 * recall * precision / (recall + precision + np.finfo(np.float32).eps)
    return f1


def cm2score(confusion_matrix):
    hist = confusion_matrix
    tp = hist[1, 1]
    fn = hist[1, 0]
    fp = hist[0, 1]
    tn = hist[0, 0]
    # acc
    oa = (tp + tn) / (tp + fn + fp + tn + np.finfo(np.float32).eps)
    # recall
    recall = tp / (tp + fn + np.finfo(np.float32).eps)
    # precision
    precision = tp / (tp + fp + np.finfo(np.float32).eps)
    # F1 score
    f1 = 2 * recall * precision / (recall + precision + np.finfo(np.float32).eps)
    # IoU
    iou = tp / (tp + fp + fn + np.finfo(np.float32).eps)
    # pre
    pre = ((tp + fn) * (tp + fp) + (tn + fp) * (tn + fn)) / (tp + fp + tn + fn) ** 2
    # kappa
    kappa = (oa - pre) / (1 - pre)
    score_dict = {'Kappa': kappa, 'IoU': iou, 'F1': f1, 'OA': oa, 'recall': recall, 'precision': precision, 'Pre': pre}
    return score_dict


def get_confuse_matrix(num_classes, label_gts, label_preds):
    """计算一组预测的混淆矩阵"""

    def __fast_hist(label_gt, label_pred):
        """
        Collect values for Confusion Matrix
        For reference, please see: https://en.wikipedia.org/wiki/Confusion_matrix
        :param label_gt: <np.array> ground-truth
        :param label_pred: <np.array> prediction
        :return: <np.ndarray> values for confusion matrix
        """
        mask = (label_gt >= 0) & (label_gt < num_classes)
        hist = np.bincount(num_classes * label_gt[mask].astype(int) + label_pred[mask],
                           minlength=num_classes ** 2).reshape(num_classes, num_classes)
        return hist

    confusion_matrix = np.zeros((num_classes, num_classes))
    for lt, lp in zip(label_gts, label_preds):
        confusion_matrix += __fast_hist(lt.flatten(), lp.flatten())
    return confusion_matrix

import torch.nn as nn
import torch


class ConvBNReLU(nn.Sequential):
    def __init__(self, in_planes, out_planes, kernel_size=3, stride=1, groups=1, dilation=1):
        padding = (kernel_size - 1) // 2
        if dilation != 1:
            padding = dilation
        super(ConvBNReLU, self).__init__(
            nn.Conv2d(in_planes, out_planes, kernel_size, stride, padding, groups=groups, dilation=dilation,
                      bias=False),
            nn.BatchNorm2d(out_planes),
            nn.ReLU6(inplace=True)
        )


class InvertedResidual(nn.Module):
    def __init__(self, inp, oup, stride, expand_ratio, dilation=1):
        super(InvertedResidual, self).__init__()
        self.stride = stride
        assert stride in [1, 2]

        hidden_dim = int(round(inp * expand_ratio))
        self.use_res_connect = self.stride == 1 and inp == oup

        layers = []
        if expand_ratio != 1:
            # pw
            layers.append(ConvBNReLU(inp, hidden_dim, kernel_size=1))
        layers.extend([
            # dw
            ConvBNReLU(hidden_dim, hidden_dim, stride=stride, groups=hidden_dim, dilation=dilation),
            # pw-linear
            nn.Conv2d(hidden_dim, oup, 1, 1, 0, bias=False),
            nn.BatchNorm2d(oup),
        ])
        self.conv = nn.Sequential(*layers)

    def forward(self, x):
        if self.use_res_connect:
            return x + self.conv(x)
        else:
            return self.conv(x)


class MobileNetV2(nn.Module):
    def __init__(self, pretrained=None, num_classes=1000, width_mult=1.0):
        super(MobileNetV2, self).__init__()
        block = InvertedResidual
        input_channel = 32
        inverted_residual_setting = [
            # t, c, n, s, d
            [1, 16, 1, 1, 1],
            [6, 24, 2, 2, 1],
            [6, 32, 3, 2, 1],
            [6, 64, 4, 2, 1],
            [6, 96, 3, 1, 1],
            [6, 160, 3, 2, 1],
            [6, 320, 1, 1, 1],
        ]

        # building first layer
        input_channel = int(input_channel * width_mult)
        features = [ConvBNReLU(3, input_channel, stride=2)]
        # building inverted residual blocks
        for t, c, n, s, d in inverted_residual_setting:
            output_channel = int(c * width_mult)
            for i in range(n):
                stride = s if i == 0 else 1
                dilation = d if i == 0 else 1
                features.append(block(input_channel, output_channel, stride, expand_ratio=t, dilation=d))
                input_channel = output_channel
        # make it nn.Sequential
        self.features = nn.Sequential(*features)

        # weight initialization
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode='fan_out')
                if m.bias is not None:
                    nn.init.zeros_(m.bias)
            elif isinstance(m, nn.BatchNorm2d):
                nn.init.ones_(m.weight)
                nn.init.zeros_(m.bias)
            elif isinstance(m, nn.Linear):
                nn.init.normal_(m.weight, 0, 0.01)
                nn.init.zeros_(m.bias)

    def forward(self, x):
        res = []
        for idx, m in enumerate(self.features):
            x = m(x)
            if idx in [1, 3, 6, 13, 17]:
                res.append(x)
        return res


def mobilenet_v2(pretrained_path=None):
    model = MobileNetV2()
    if pretrained_path:
        state = torch.load(pretrained_path, map_location="cpu", weights_only=True)
        missing, unexpected = model.load_state_dict(state, strict=False)
        if missing:
            raise RuntimeError(f"MobileNetV2 missing pretrained tensors: {missing}")
        print(f"[PRETRAIN-LOAD] MobileNetV2 tensors={len(model.state_dict())} missing=0 unexpected={unexpected}", flush=True)
    return model

"""Top-down DCR decoder with foldable local refinement.

TAR features at 8/16/32/64 spatial resolution are fused from coarse to fine.
The four decoder outputs support deep supervision at input resolution.
"""
import torch.nn as nn
import torch.nn.functional as F

from model.str_reparam import (
    RepDW3,
    RepPW1x1,
    RepPairFuse1x1,
    switch_module_to_deploy,
)


class RepLocalBlock(nn.Module):
    """Local refinement block: DW3 (residual) -> SiLU -> PW1 (residual) -> SiLU."""

    def __init__(self, dim, use_aux=True, use_residual=True, deploy=False):
        super().__init__()
        self.dim = dim
        self.dw = RepDW3(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.pw = RepPW1x1(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.act = nn.SiLU()

    def forward(self, x):
        x = self.act(self.dw(x))
        x = self.act(self.pw(x))
        return x

    def switch_to_deploy(self):
        self.dw.switch_to_deploy()
        self.pw.switch_to_deploy()
        return self


class DCRDecoder(nn.Module):
    """Top-down multi-scale decoder over TAR features [t1, t2, t3, t4] (dim ch)."""

    def __init__(self, dim=64, use_aux=True, use_residual=True, deploy=False):
        super().__init__()
        self.dim = dim
        self.use_aux = use_aux
        self.act = nn.SiLU()

        self.fuse3 = RepPairFuse1x1(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.block3 = RepLocalBlock(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.fuse2 = RepPairFuse1x1(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.block2 = RepLocalBlock(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.fuse1 = RepPairFuse1x1(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.block1 = RepLocalBlock(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)
        self.refine = RepLocalBlock(dim, use_aux=use_aux, use_residual=use_residual, deploy=deploy)

    def forward(self, feats):
        return self.forward_pyramid(feats)[0]

    def forward_pyramid(self, feats):
        """Return the four decoder scales for deep supervision."""
        t1, t2, t3, t4 = feats

        d4 = t4

        u4 = F.interpolate(d4, size=t3.shape[-2:], mode="bilinear", align_corners=False)
        d3 = self.block3(self.act(self.fuse3(t3, u4)))

        u3 = F.interpolate(d3, size=t2.shape[-2:], mode="bilinear", align_corners=False)
        d2 = self.block2(self.act(self.fuse2(t2, u3)))

        u2 = F.interpolate(d2, size=t1.shape[-2:], mode="bilinear", align_corners=False)
        d1 = self.block1(self.act(self.fuse1(t1, u2)))

        d1 = self.refine(d1)
        return [d1, d2, d3, d4]

    def switch_to_deploy(self):
        switch_module_to_deploy(self)
        return self

"""Foldable depthwise, pointwise and paired-feature convolution blocks.

Training branches use separate BatchNorm layers and learnable residuals.
Kernel and bias composition uses FP64 before one FP32 cast for deployment.
"""
import torch
import torch.nn as nn
import torch.nn.functional as F


def fold_conv_bn(weight, bias, bn):
    """Fold one Conv (weight, bias) + BatchNorm into (W', b').

    Returns float64 tensors: ALL branch composition must stay in float64 and
    cast to FP32 once, at the final deploy-conv assignment.
    """
    C = weight.shape[0]
    eps = bn.eps if bn.eps is not None else 1e-5
    gamma = bn.weight.detach().double()
    beta = bn.bias.detach().double()
    mu = bn.running_mean.detach().double()
    var = bn.running_var.detach().double()
    std = torch.sqrt(var + eps)
    t = gamma / std
    bd = bias.detach().double() if bias is not None else torch.zeros(C, dtype=torch.float64, device=weight.device)
    W = weight.detach().double() * t.view(C, 1, 1, 1)
    b = beta + (bd - mu) * t
    return W, b


def pad_1x3_to_3x3(w):
    return F.pad(w, (0, 0, 1, 1))


def pad_3x1_to_3x3(w):
    return F.pad(w, (1, 1, 0, 0))


def identity_dw_kernel(channels, k=3, device=None, dtype=torch.float32):
    kernel = torch.zeros(channels, 1, k, k, device=device, dtype=dtype)
    kernel[:, 0, k // 2, k // 2] = 1.0
    return kernel


def identity_1x1_kernel(channels, device=None, dtype=torch.float32):
    return torch.eye(channels, device=device, dtype=dtype).view(channels, channels, 1, 1)


class RepDW3(nn.Module):
    """Depthwise Rep 3x3: DW3x3 + DW1x3 + DW3x1 (each with own BN) + alpha*x -> single DW3x3."""

    def __init__(self, channels, use_aux=True, use_residual=True, deploy=False):
        super().__init__()
        self.channels = channels
        self.use_aux = use_aux
        self.use_residual = use_residual
        self.deploy = deploy
        if deploy:
            self.dw = nn.Conv2d(channels, channels, 3, 1, 1, groups=channels, bias=True)
        else:
            self.dw3 = nn.Conv2d(channels, channels, 3, 1, 1, groups=channels, bias=False)
            self.bn3 = nn.BatchNorm2d(channels)
            nn.init.kaiming_normal_(self.dw3.weight, mode="fan_out", nonlinearity="relu")
            if self.use_aux:
                # Keep auxiliary construction independent of the shared initialization stream.
                from torch.nn.utils import skip_init
                self.dw13 = skip_init(nn.Conv2d, channels, channels, (1, 3), 1, (0, 1), groups=channels, bias=False)
                self.bn13 = nn.BatchNorm2d(channels)
                self.dw31 = skip_init(nn.Conv2d, channels, channels, (3, 1), 1, (1, 0), groups=channels, bias=False)
                self.bn31 = nn.BatchNorm2d(channels)
                nn.init.zeros_(self.dw13.weight)
                nn.init.zeros_(self.dw31.weight)
            if self.use_residual:
                self.alpha = nn.Parameter(torch.ones(1))

    def forward(self, x):
        if self.deploy:
            return self.dw(x)
        y = self.bn3(self.dw3(x))
        if self.use_aux:
            y = y + self.bn13(self.dw13(x)) + self.bn31(self.dw31(x))
        if self.use_residual:
            y = y + self.alpha * x
        return y

    def get_equivalent_kernel_bias(self):
        W3, b3 = fold_conv_bn(self.dw3.weight, None, self.bn3)
        k = W3
        b = b3
        if self.use_aux:
            W13, b13 = fold_conv_bn(self.dw13.weight, None, self.bn13)
            W31, b31 = fold_conv_bn(self.dw31.weight, None, self.bn31)
            k = k + pad_1x3_to_3x3(W13) + pad_3x1_to_3x3(W31)
            b = b + b13 + b31
        if self.use_residual:
            k = k + self.alpha.detach().double() * identity_dw_kernel(
                self.channels, 3, device=k.device, dtype=torch.float64)
        return k, b

    def branch_stats(self):
        s = {"dw3": self.dw3.weight.norm().item()}
        if self.use_aux:
            s["dw13"] = self.dw13.weight.norm().item()
            s["dw31"] = self.dw31.weight.norm().item()
        if self.use_residual:
            s["alpha"] = self.alpha.item()
        return s

    def switch_to_deploy(self):
        if self.deploy:
            return self
        kernel, bias = self.get_equivalent_kernel_bias()
        self.dw = nn.Conv2d(self.channels, self.channels, 3, 1, 1, groups=self.channels, bias=True)
        self.dw.weight.data = kernel.float()
        self.dw.bias.data = bias.float()
        self.deploy = True
        for name in ("dw13", "dw31", "bn3", "bn13", "bn31", "dw3", "alpha"):
            if hasattr(self, name):
                delattr(self, name)
        return self


class RepPW1x1(nn.Module):
    """Pointwise Rep 1x1: main + serial low-rank + diag (each BN) + alpha*x -> single 1x1."""

    def __init__(self, channels, r=None, use_aux=True, use_residual=True, deploy=False):
        super().__init__()
        self.channels = channels
        self.r = r if r is not None else max(1, channels // 4)
        self.use_aux = use_aux
        self.use_residual = use_residual
        self.deploy = deploy
        if deploy:
            self.pw = nn.Conv2d(channels, channels, 1, bias=True)
        else:
            self.pw_main = nn.Conv2d(channels, channels, 1, bias=False)
            self.bn_main = nn.BatchNorm2d(channels)
            nn.init.kaiming_normal_(self.pw_main.weight, mode="fan_out", nonlinearity="relu")
            if self.use_aux:
                # Keep auxiliary construction independent of the shared initialization stream.
                from torch.nn.utils import skip_init
                self.pw1 = skip_init(nn.Conv2d, channels, self.r, 1, bias=False)
                self.pw2 = skip_init(nn.Conv2d, self.r, channels, 1, bias=False)
                self.bn_lr = nn.BatchNorm2d(channels)
                self.diag = nn.Parameter(torch.zeros(channels))
                _g = torch.Generator().manual_seed(0x5F1)
                nn.init.kaiming_normal_(self.pw1.weight, mode="fan_out", nonlinearity="relu", generator=_g)
                nn.init.normal_(self.pw2.weight, mean=0.0, std=1e-3, generator=_g)
                # Zero-initialized normalization keeps the serial branch inactive at initialization.
                nn.init.zeros_(self.bn_lr.weight)
                nn.init.zeros_(self.bn_lr.bias)
            if self.use_residual:
                self.alpha = nn.Parameter(torch.ones(1))

    def forward(self, x):
        if self.deploy:
            return self.pw(x)
        y = self.bn_main(self.pw_main(x))
        if self.use_aux:
            y = y + self.bn_lr(self.pw2(self.pw1(x))) + self.diag.view(1, -1, 1, 1) * x
        if self.use_residual:
            y = y + self.alpha * x
        return y

    def get_equivalent_kernel_bias(self):
        C = self.channels
        W, b = fold_conv_bn(self.pw_main.weight, None, self.bn_main)
        if self.use_aux:
            W2 = self.pw2.weight[:, :, 0, 0].double()  # (C, r)
            W1 = self.pw1.weight[:, :, 0, 0].double()  # (r, C)
            W_serial = (W2 @ W1).view(C, C, 1, 1)      # float64 compose
            W_lr, b_lr = fold_conv_bn(W_serial, None, self.bn_lr)
            D = torch.diag(self.diag.detach().double()).view(C, C, 1, 1)
            W = W + W_lr + D
            b = b + b_lr
        if self.use_residual:
            W = W + self.alpha.detach().double() * identity_1x1_kernel(
                C, device=W.device, dtype=torch.float64)
        return W, b

    def branch_stats(self):
        s = {"main": self.pw_main.weight.norm().item()}
        if self.use_aux:
            s["pw1"] = self.pw1.weight.norm().item()
            s["pw2"] = self.pw2.weight.norm().item()
        if self.use_residual:
            s["alpha"] = self.alpha.item()
        return s

    def switch_to_deploy(self):
        if self.deploy:
            return self
        kernel, bias = self.get_equivalent_kernel_bias()
        self.pw = nn.Conv2d(self.channels, self.channels, 1, bias=True)
        self.pw.weight.data = kernel.float()
        self.pw.bias.data = bias.float()
        self.deploy = True
        for name in ("pw1", "pw2", "diag", "bn_main", "bn_lr", "pw_main", "alpha"):
            if hasattr(self, name):
                delattr(self, name)
        return self


class RepPairFuse1x1(nn.Module):
    """Cross-scale fusion Rep 1x1: concat + sum + diff (each BN) + alpha*L -> single 1x1."""

    def __init__(self, channels, use_aux=True, use_residual=True, deploy=False):
        super().__init__()
        self.channels = channels
        self.use_aux = use_aux
        self.use_residual = use_residual
        self.deploy = deploy
        if deploy:
            self.fuse = nn.Conv2d(2 * channels, channels, 1, bias=True)
        else:
            self.fuse_c = nn.Conv2d(2 * channels, channels, 1, bias=False)
            self.bn_c = nn.BatchNorm2d(channels)
            nn.init.kaiming_normal_(self.fuse_c.weight, mode="fan_out", nonlinearity="relu")
            if self.use_aux:
                # Keep auxiliary construction independent of the shared initialization stream.
                from torch.nn.utils import skip_init
                self.fuse_s = skip_init(nn.Conv2d, channels, channels, 1, bias=False)
                self.bn_s = nn.BatchNorm2d(channels)
                self.fuse_d = skip_init(nn.Conv2d, channels, channels, 1, bias=False)
                self.bn_d = nn.BatchNorm2d(channels)
                nn.init.zeros_(self.fuse_s.weight)
                nn.init.zeros_(self.fuse_d.weight)
            if self.use_residual:
                self.alpha = nn.Parameter(torch.ones(1))

    def forward(self, L, H):
        if self.deploy:
            return self.fuse(torch.cat([L, H], dim=1))
        y = self.bn_c(self.fuse_c(torch.cat([L, H], dim=1)))
        if self.use_aux:
            y = y + self.bn_s(self.fuse_s(L + H)) + self.bn_d(self.fuse_d(H - L))
        if self.use_residual:
            y = y + self.alpha * L
        return y

    def get_equivalent_kernel_bias(self):
        C = self.channels
        W_c, b_c = fold_conv_bn(self.fuse_c.weight, None, self.bn_c)
        W_cL = W_c[:, :C]
        W_cH = W_c[:, C:]
        if self.use_aux:
            W_s, b_s = fold_conv_bn(self.fuse_s.weight, None, self.bn_s)
            W_d, b_d = fold_conv_bn(self.fuse_d.weight, None, self.bn_d)
            W_L = W_cL + W_s - W_d
            W_H = W_cH + W_s + W_d
            b = b_c + b_s + b_d
        else:
            W_L, W_H = W_cL, W_cH
            b = b_c
        if self.use_residual:
            W_L = W_L + self.alpha.detach().double() * identity_1x1_kernel(
                C, device=W_L.device, dtype=torch.float64)
        W = torch.cat([W_L, W_H], dim=1)
        return W, b

    def branch_stats(self):
        s = {"concat": self.fuse_c.weight.norm().item()}
        if self.use_aux:
            s["sum"] = self.fuse_s.weight.norm().item()
            s["diff"] = self.fuse_d.weight.norm().item()
        if self.use_residual:
            s["alpha"] = self.alpha.item()
        return s

    def switch_to_deploy(self):
        if self.deploy:
            return self
        kernel, bias = self.get_equivalent_kernel_bias()
        self.fuse = nn.Conv2d(2 * self.channels, self.channels, 1, bias=True)
        self.fuse.weight.data = kernel.float()
        self.fuse.bias.data = bias.float()
        self.deploy = True
        for name in ("fuse_s", "fuse_d", "bn_c", "bn_s", "bn_d", "fuse_c", "alpha"):
            if hasattr(self, name):
                delattr(self, name)
        return self


def switch_module_to_deploy(module):
    for m in module.modules():
        if m is not module and hasattr(m, "switch_to_deploy"):
            m.switch_to_deploy()
    return module

"""Four-scale temporal fusion with foldable concat, sum and signed-difference branches."""
import torch
import torch.nn as nn

from model.str_reparam import fold_conv_bn, switch_module_to_deploy
from model.str_dcr import RepLocalBlock


class TemporalRep1x1(nn.Module):
    """Bi-temporal 1x1 re-parameterization: concat + sum + signed-diff (each BN) -> single 1x1."""

    def __init__(self, in_channels, out_channels, use_aux=True, deploy=False):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.use_aux = use_aux
        self.deploy = deploy
        if deploy:
            self.proj = nn.Conv2d(2 * in_channels, out_channels, 1, bias=True)
        else:
            self.proj_c = nn.Conv2d(2 * in_channels, out_channels, 1, bias=False)
            self.bn_c = nn.BatchNorm2d(out_channels)
            nn.init.kaiming_normal_(self.proj_c.weight, mode="fan_out", nonlinearity="relu")
            if self.use_aux:
                # Keep auxiliary construction independent of the shared initialization stream.
                from torch.nn.utils import skip_init
                self.proj_s = skip_init(nn.Conv2d, in_channels, out_channels, 1, bias=False)
                self.bn_s = nn.BatchNorm2d(out_channels)
                self.proj_d = skip_init(nn.Conv2d, in_channels, out_channels, 1, bias=False)
                self.bn_d = nn.BatchNorm2d(out_channels)
                nn.init.zeros_(self.proj_s.weight)
                nn.init.zeros_(self.proj_d.weight)

    def forward(self, P, Q):
        if self.deploy:
            return self.proj(torch.cat([P, Q], dim=1))
        y = self.bn_c(self.proj_c(torch.cat([P, Q], dim=1)))
        if self.use_aux:
            y = y + self.bn_s(self.proj_s(P + Q)) + self.bn_d(self.proj_d(Q - P))
        return y

    def get_equivalent_kernel_bias(self):
        C = self.in_channels
        W_c, b_c = fold_conv_bn(self.proj_c.weight, None, self.bn_c)
        W_P = W_c[:, :C]
        W_Q = W_c[:, C:]
        b = b_c
        if self.use_aux:
            W_s, b_s = fold_conv_bn(self.proj_s.weight, None, self.bn_s)
            W_d, b_d = fold_conv_bn(self.proj_d.weight, None, self.bn_d)
            W_P = W_P + W_s - W_d
            W_Q = W_Q + W_s + W_d
            b = b + b_s + b_d
        W = torch.cat([W_P, W_Q], dim=1)
        return W, b

    def branch_stats(self):
        s = {"concat": self.proj_c.weight.norm().item()}
        if self.use_aux:
            s["sum"] = self.proj_s.weight.norm().item()
            s["diff"] = self.proj_d.weight.norm().item()
        return s

    def switch_to_deploy(self):
        if self.deploy:
            return self
        kernel, bias = self.get_equivalent_kernel_bias()
        self.proj = nn.Conv2d(2 * self.in_channels, self.out_channels, 1, bias=True)
        self.proj.weight.data = kernel.float()
        self.proj.bias.data = bias.float()
        self.deploy = True
        for name in ("proj_s", "proj_d", "bn_c", "bn_s", "bn_d", "proj_c"):
            if hasattr(self, name):
                delattr(self, name)
        return self


class TARStage(nn.Module):
    """One scale: temporal 1x1 projection, SiLU and local refinement."""

    def __init__(self, in_channels, dim, use_temporal_aux=True, use_dcr_aux=True,
                 use_residual=True, deploy=False):
        super().__init__()
        self.temporal = TemporalRep1x1(in_channels, dim, use_aux=use_temporal_aux, deploy=deploy)
        self.block = RepLocalBlock(dim, use_aux=use_dcr_aux, use_residual=use_residual, deploy=deploy)
        self.act = nn.SiLU()

    def forward(self, P, Q):
        x = self.act(self.temporal(P, Q))
        return self.block(x)

    def switch_to_deploy(self):
        self.temporal.switch_to_deploy()
        self.block.switch_to_deploy()
        return self


class MultiScaleTAR(nn.Module):
    """Four-scale temporal fusion from paired encoder features to decoder inputs.

    Input scale order matches STR convention: stage1 (finest, 64x64) .. stage4
    (coarsest, 8x8); output list [t1, t2, t3, t4] feeds DCRDecoder.
    """

    def __init__(self, encoder_dims=(64, 64, 64, 64), dim=64,
                 use_temporal_aux=True, use_dcr_aux=True, use_residual=True, deploy=False):
        super().__init__()
        self.stage1 = TARStage(encoder_dims[0], dim, use_temporal_aux, use_dcr_aux,
                               use_residual, deploy)
        self.stage2 = TARStage(encoder_dims[1], dim, use_temporal_aux, use_dcr_aux,
                               use_residual, deploy)
        self.stage3 = TARStage(encoder_dims[2], dim, use_temporal_aux, use_dcr_aux,
                               use_residual, deploy)
        self.stage4 = TARStage(encoder_dims[3], dim, use_temporal_aux, use_dcr_aux,
                               use_residual, deploy)

    def forward(self, pre_feats, post_feats):
        t1 = self.stage1(pre_feats[0], post_feats[0])
        t2 = self.stage2(pre_feats[1], post_feats[1])
        t3 = self.stage3(pre_feats[2], post_feats[2])
        t4 = self.stage4(pre_feats[3], post_feats[3])
        return [t1, t2, t3, t4]

    def switch_to_deploy(self):
        switch_module_to_deploy(self)
        return self

"""Curvature -1 paired geometry in Lorentz coordinates (time coordinate first)."""
import torch


class _DirectPairwiseDistance(torch.autograd.Function):
    """Direct Euclidean norms with a matrix-based gradient requiring no N×M×D buffer."""

    @staticmethod
    def forward(ctx, x, y):
        squared = (x[..., 0, None] - y[..., None, :, 0]).square()
        for dim in range(1, x.shape[-1]):
            squared.add_((x[..., dim, None] - y[..., None, :, dim]).square())
        distance = squared.sqrt_()
        ctx.save_for_backward(x, y, distance)
        return distance

    @staticmethod
    def backward(ctx, grad_output):
        x, y, distance = ctx.saved_tensors
        weights = grad_output / distance.clamp_min(torch.finfo(distance.dtype).tiny)
        weights = torch.where(distance > 0, weights, 0.)
        grad_x = x * weights.sum(-1, keepdim=True) - weights @ y
        grad_y = y * weights.sum(-2).unsqueeze(-1) - weights.transpose(-1, -2) @ x
        return grad_x.sum_to_size(x.shape), grad_y.sum_to_size(y.shape)


def euclidean_pairwise_distance(x, y):
    """Exact pairwise Euclidean norms; use the zero subgradient at coincident points."""
    return _DirectPairwiseDistance.apply(x, y)


def lorentz_inner(x, y, keepdim=False):
    """Lorentz inner product with arbitrary broadcastable leading dimensions."""
    value = (x[..., 1:] * y[..., 1:]).sum(-1) - x[..., 0] * y[..., 0]
    return value.unsqueeze(-1) if keepdim else value


def lorentz_normalize(x):
    """Normalize a positive weighted sum of upper-sheet hyperboloid points."""
    norm = (-lorentz_inner(x, x, keepdim=True)).clamp_min(1e-8).sqrt()
    return x / norm


def lorentz_midpoint(x, y):
    """Exact equal-weight two-point Frechet mean, without an iterative solver."""
    delta = y - x
    # -<x+y,x+y> = 4 + <y-x,y-x> for unit hyperboloid points.
    denominator = (4 + lorentz_inner(delta, delta, keepdim=True).clamp_min(0)).sqrt()
    return (x + y) / denominator


def lorentz_logmap(base, target):
    """Log_base(target), with the analytic coincident-point coefficient limit."""
    delta = target - base
    # z = -<base,target>-1; this form avoids subtracting nearly equal products.
    z = (0.5 * lorentz_inner(delta, delta, keepdim=True)).clamp_min(0)
    regular_z = z.clamp_min(1e-3)
    regular = (2 * torch.asinh((regular_z / 2).sqrt()) /
               (regular_z * (regular_z + 2)).sqrt())
    series = 1 - z / 3 + 2 * z.square() / 15 - 2 * z.pow(3) / 35
    coefficient = torch.where(z < 1e-3, series, regular)
    return coefficient * (delta - z * base)


def parallel_transport_to_origin(base, tangent):
    """Transport an ambient Lorentz tangent to the origin; return spatial coordinates."""
    return tangent[..., 1:] - (tangent[..., :1] / (base[..., :1] + 1)) * base[..., 1:]


def pair_midpoint_difference(x, y):
    """Return midpoint m and spatial PT_{m->o}(Log_m(y)); swap negates the latter."""
    midpoint = lorentz_midpoint(x, y)
    half_difference = (y - x) * 0.5
    squared = lorentz_inner(half_difference, half_difference, keepdim=True).clamp_min(0)
    regular_squared = squared.clamp_min(1e-3)
    root = regular_squared.sqrt()
    regular = torch.asinh(root) / root
    series = 1 - squared / 6 + 3 * squared.square() / 40 - 5 * squared.pow(3) / 112
    coefficient = torch.where(squared < 1e-3, series, regular)
    tangent = coefficient * half_difference
    return midpoint, parallel_transport_to_origin(midpoint, tangent)


def pairwise_lorentz_distance(x, y):
    """All-pairs geodesic distances, with finite gradients for coincident points.

    The equivalent Poincare asinh identity avoids cancellation in acosh(-<x,y>)
    for nearby large Lorentz coordinates. Direct differences keep exact zero distances
    without materializing a query-by-key-by-channel tensor.
    """
    x_time, y_time = x[..., :1] + 1, y[..., :1] + 1
    x_ball, y_ball = x[..., 1:] / x_time, y[..., 1:] / y_time
    chord = euclidean_pairwise_distance(x_ball, y_ball)
    factor = (x_time * y_time.transpose(-1, -2)).sqrt() * 0.5
    return 2 * torch.asinh(chord * factor)

"""Dense paired differences and separate regional mean/contrast retrieval."""
import math

import torch
from torch import nn

from model.layers.pyramid_context import (
    attention_diagnostics, binomial_highpass, channel_norm, feature_regions,
    merge_heads, project_tokens, residual_diagnostics, source_bias, split_heads,
)


class PairedEuclideanContext(nn.Module):
    geometry = "euclidean"
    value_geometry = "euclidean"

    def __init__(self, in_channels=(24, 32, 96, 320), channels=64,
                 heads=4, grid_sizes=(8, 4, 4, 4)):
        super().__init__()
        self.heads = heads
        self.grid_sizes = tuple(grid_sizes)
        self.regions_per_scale = tuple(grid ** 2 for grid in self.grid_sizes)
        self.tokens_per_scale = tuple(2 * count for count in self.regions_per_scale)
        self.norm = nn.LayerNorm(channels)
        self.guide = nn.Conv2d(channels, 1, 1)
        self.input_norm = nn.ModuleList(nn.LayerNorm(c) for c in in_channels)
        self.pair_proj = nn.ModuleList(nn.Conv2d(c, 32, 1, bias=False) for c in in_channels)
        self.diff_proj = nn.Conv2d(32, channels, 1, bias=False)
        self.q = nn.Conv2d(channels, 64, 1, bias=False)
        self.k_mean = nn.Conv2d(channels, 64, 1, bias=False)
        self.k_contrast = nn.Conv2d(channels, 64, 1, bias=False)
        self.v_mean = nn.Conv2d(channels, 32, 1, bias=False)
        self.v_contrast = nn.Conv2d(channels, 32, 1, bias=False)
        self.out = nn.Conv2d(64, channels, 1, bias=False)
        self.scale_bias = nn.Parameter(torch.zeros(2, heads, 4, 4))
        self.gamma_context = nn.Parameter(torch.zeros(4))
        self.gamma_detail = nn.Parameter(torch.zeros(2))
        self.boundary = nn.Conv2d(channels + 64, 1, 1)
        self.collect_diag = False
        self.last_diag = {}

    def paired_features(self, features_a, features_b, tar):
        differences, normalized, logits = [], [], []
        for a, b, t, norm, projection in zip(features_a, features_b, tar, self.input_norm, self.pair_proj):
            difference = (projection(channel_norm(norm, b)) - projection(channel_norm(norm, a))).abs()
            feature = channel_norm(self.norm, t + self.diff_proj(difference))
            differences.append(difference)
            normalized.append(feature)
            logits.append(self.guide(feature))
        return differences, normalized, logits

    @staticmethod
    def prototypes(feature, probability, grid):
        cells = feature_regions(feature, grid)
        membership = feature_regions(probability, grid)
        mean = cells.mean(-2)
        changed = ((membership * cells).sum(-2) + mean) / (membership.sum(-2) + 1.)
        unchanged = (((1 - membership) * cells).sum(-2) + mean) / ((1 - membership).sum(-2) + 1.)
        return mean, changed - unchanged

    def attention(self, query, key, query_scale, bank):
        logits = query @ key.transpose(-1, -2) / math.sqrt(query.shape[-1])
        return (logits + source_bias(self.scale_bias[bank], query_scale, self.regions_per_scale)).softmax(-1)

    def forward(self, features_a, features_b, tar):
        differences, normalized, guide = self.paired_features(features_a, features_b, tar)
        prototypes = [self.prototypes(x, g.sigmoid(), grid)
                      for x, g, grid in zip(normalized, guide, self.grid_sizes)]
        mean, contrast = (torch.cat(bank, dim=1) for bank in zip(*prototypes))
        keys = (project_tokens(self.k_mean, mean, self.heads),
                project_tokens(self.k_contrast, contrast, self.heads))
        values = (project_tokens(self.v_mean, mean, self.heads),
                  project_tokens(self.v_contrast, contrast, self.heads))
        enhanced, boundaries, diagnostics = [], [], {}
        for index, (original, feature, difference) in enumerate(zip(tar, normalized, differences)):
            height, width = feature.shape[-2:]
            query = split_heads(self.q(feature), self.heads).flatten(2, 3)
            probabilities = [self.attention(query, key, index, bank) for bank, key in enumerate(keys)]
            content = [merge_heads(probability @ value, height, width)
                       for probability, value in zip(probabilities, values)]
            update = self.out(torch.cat(content, dim=1))
            result = original + self.gamma_context[index] * update
            local, detail_gamma = None, 0.
            if index < 2:
                high = binomial_highpass(difference)
                boundary = self.boundary(torch.cat((feature, difference, high), dim=1))
                local = self.diff_proj(difference + boundary.sigmoid() * high)
                detail_gamma = self.gamma_detail[index]
                result = result + detail_gamma * local
                boundaries.append(boundary)
            enhanced.append(result)
            if self.collect_diag:
                with torch.no_grad():
                    diag = residual_diagnostics(original, update, local, self.gamma_context[index], detail_gamma)
                    diag.update(queries=height * width, contexts=mean.shape[1] + contrast.shape[1],
                                contexts_per_bank=mean.shape[1], guide_mean=float(guide[index].sigmoid().mean()),
                                mean_content_rms=float(content[0].square().mean().sqrt()),
                                contrast_content_rms=float(content[1].square().mean().sqrt()),
                                k_mean_weight_rms=float(self.k_mean.weight.square().mean().sqrt()),
                                k_contrast_weight_rms=float(self.k_contrast.weight.square().mean().sqrt()),
                                v_mean_weight_rms=float(self.v_mean.weight.square().mean().sqrt()),
                                v_contrast_weight_rms=float(self.v_contrast.weight.square().mean().sqrt()),
                                dense_difference_rms=float(difference.square().mean().sqrt()),
                                mean_prototype_rms=float(prototypes[index][0].square().mean().sqrt()),
                                contrast_prototype_rms=float(prototypes[index][1].square().mean().sqrt()))
                    for bank, probability in zip(("mean", "contrast"), probabilities):
                        diag.update({f"{bank}_{key}": value for key, value in
                                     attention_diagnostics(probability, self.regions_per_scale).items()})
                    if index < 2:
                        diag.update(boundary_mean=float(boundary.sigmoid().mean()),
                                    detail_highpass_rms=float(high.square().mean().sqrt()))
                    diagnostics[f"stage{index + 1}"] = diag
        if self.collect_diag:
            self.last_diag = diagnostics
        return enhanced, dict(guide_logits=tuple(guide), boundary_logits=tuple(boundaries))

"""Paired hyperbolic midpoint and common-tangent difference retrieval."""
import math

import torch
from torch import nn
from torch.nn import functional as F

from model.layers.lorentz_pair_ops import (
    euclidean_pairwise_distance, lorentz_normalize, pair_midpoint_difference, pairwise_lorentz_distance,
)
from model.layers.poincare_ops import RADIUS, expmap0, to_lorentz
from model.layers.pyramid_context import (
    attention_diagnostics, binomial_highpass, channel_norm, feature_regions, merge_heads,
    project_tokens, regions, residual_diagnostics, source_bias, split_heads,
)


class PairedHyperbolicContext(nn.Module):
    geometry = "hyperbolic"
    value_geometry = "euclidean"

    def __init__(self, in_channels=(24, 32, 96, 320), channels=64,
                 heads=4, grid_sizes=(8, 4, 4, 4)):
        super().__init__()
        self.heads = heads
        self.grid_sizes = tuple(grid_sizes)
        self.tokens_per_scale = tuple(2 * grid ** 2 for grid in self.grid_sizes)
        self.norm = nn.LayerNorm(channels)
        self.guide = nn.Conv2d(channels, 1, 1)
        self.input_norm = nn.ModuleList(nn.LayerNorm(c) for c in in_channels)
        self.pair_proj = nn.ModuleList(nn.Conv2d(c, 32, 1, bias=False) for c in in_channels)
        self.diff_proj = nn.Conv2d(32, channels, 1, bias=False)
        self.v = nn.Conv2d(channels, 64, 1, bias=False)
        self.out = nn.Conv2d(64, channels, 1, bias=False)
        self.scale_bias = nn.Parameter(torch.zeros(heads, 4, 4))
        self.raw_tau_mid = nn.Parameter(torch.full((heads,), math.log(math.expm1(.2))))
        self.raw_tau_diff = nn.Parameter(torch.full((heads,), math.log(math.expm1(.2))))
        self.gamma_context = nn.Parameter(torch.zeros(4))
        self.gamma_detail = nn.Parameter(torch.zeros(2))
        self.boundary = nn.Conv2d(channels + 64, 1, 1)
        self.collect_diag = False
        self.last_diag = {}

    def paired_features(self, features_a, features_b, tar):
        midpoints, signed, magnitude, normalized, logits, clip_fractions = [], [], [], [], [], []
        for a, b, t, norm, projection in zip(features_a, features_b, tar, self.input_norm, self.pair_proj):
            tangent_a = split_heads(projection(channel_norm(norm, a)), self.heads) / math.sqrt(8)
            tangent_b = split_heads(projection(channel_norm(norm, b)), self.heads) / math.sqrt(8)
            midpoint, difference = pair_midpoint_difference(to_lorentz(expmap0(tangent_a)),
                                                           to_lorentz(expmap0(tangent_b)))
            unsigned = difference.abs()
            height, width = t.shape[-2:]
            feature = channel_norm(self.norm, t + self.diff_proj(merge_heads(unsigned.flatten(2, 3), height, width)))
            midpoints.append(midpoint)
            signed.append(difference)
            magnitude.append(unsigned)
            normalized.append(feature)
            logits.append(self.guide(feature))
            if self.collect_diag:
                clip_fractions.append(((tangent_a.detach().norm(dim=-1) >= 2.3).float().mean()
                                       + (tangent_b.detach().norm(dim=-1) >= 2.3).float().mean()) / 2)
        return midpoints, signed, magnitude, normalized, logits, clip_fractions

    @staticmethod
    def assignments(probability, grid):
        cells = feature_regions(probability, grid).squeeze(-1)
        changed = (cells + 1e-6) / (cells.sum(-1, keepdim=True) + cells.shape[-1] * 1e-6)
        return torch.stack((torch.full_like(changed, 1 / cells.shape[-1]), changed), dim=2)

    def compress(self, midpoint, unsigned, feature, weights, grid):
        mid_cells, unsigned_cells = regions(midpoint, grid), regions(unsigned, grid)
        key_mid = lorentz_normalize(torch.einsum("brtp,bhrpd->bhrtd", weights, mid_cells)).flatten(2, 3)
        key_unsigned = torch.einsum("brtp,bhrpd->bhrtd", weights, unsigned_cells).flatten(2, 3)
        content = torch.einsum("brtp,brpc->brtc", weights, feature_regions(feature, grid)).flatten(1, 2)
        return key_mid, key_unsigned, project_tokens(self.v, content, self.heads)

    def attention(self, midpoint, unsigned, key_mid, key_unsigned, query_scale):
        midpoint_term = -pairwise_lorentz_distance(midpoint, key_mid) / (
            2 * F.softplus(self.raw_tau_mid).view(1, self.heads, 1, 1))
        difference_term = -euclidean_pairwise_distance(unsigned, key_unsigned) / F.softplus(
            self.raw_tau_diff).view(1, self.heads, 1, 1)
        probability = (midpoint_term + difference_term + source_bias(
            self.scale_bias, query_scale, self.tokens_per_scale)).softmax(-1)
        return probability, midpoint_term, difference_term

    def forward(self, features_a, features_b, tar):
        midpoint, signed, magnitude, normalized, guide, clip_fraction = self.paired_features(features_a, features_b, tar)
        weights = [self.assignments(g.sigmoid(), grid) for g, grid in zip(guide, self.grid_sizes)]
        prototypes = [self.compress(m, u, x, w, grid) for m, u, x, w, grid in
                      zip(midpoint, magnitude, normalized, weights, self.grid_sizes)]
        key_mid, key_unsigned, values = (torch.cat(bank, dim=2) for bank in zip(*prototypes))
        enhanced, boundaries, diagnostics = [], [], {}
        for index, (original, feature, mid, signed_diff, unsigned) in enumerate(
                zip(tar, normalized, midpoint, signed, magnitude)):
            height, width = original.shape[-2:]
            probability, mid_term, diff_term = self.attention(mid.flatten(2, 3), unsigned.flatten(2, 3),
                                                            key_mid, key_unsigned, index)
            update = self.out(merge_heads(probability @ values, height, width))
            result = original + self.gamma_context[index] * update
            local, detail_gamma = None, 0.
            unsigned_map = merge_heads(unsigned.flatten(2, 3), height, width)
            if index < 2:
                high = binomial_highpass(merge_heads(signed_diff.flatten(2, 3), height, width)).abs()
                boundary = self.boundary(torch.cat((feature, unsigned_map, high), dim=1))
                local = self.diff_proj(unsigned_map + boundary.sigmoid() * high)
                detail_gamma = self.gamma_detail[index]
                result = result + detail_gamma * local
                boundaries.append(boundary)
            enhanced.append(result)
            if self.collect_diag:
                with torch.no_grad():
                    radii = (mid[..., 1:] / (mid[..., :1] + 1)).norm(dim=-1)
                    diag = residual_diagnostics(original, update, local, self.gamma_context[index], detail_gamma)
                    diag.update(attention_diagnostics(probability, self.tokens_per_scale))
                    diag.update(queries=height * width, contexts=key_mid.shape[2],
                        guide_mean=float(guide[index].sigmoid().mean()),
                        temperature_mid=F.softplus(self.raw_tau_mid).tolist(),
                        temperature_diff=F.softplus(self.raw_tau_diff).tolist(),
                        midpoint_logit_mean=mid_term.mean((0, 2, 3)).tolist(),
                        midpoint_logit_std=mid_term.std((0, 2, 3), unbiased=False).tolist(),
                        difference_logit_mean=diff_term.mean((0, 2, 3)).tolist(),
                        difference_logit_std=diff_term.std((0, 2, 3), unbiased=False).tolist(),
                        midpoint_logit_key_std=mid_term.std(-1, unbiased=False).mean((0, 2)).tolist(),
                        difference_logit_key_std=diff_term.std(-1, unbiased=False).mean((0, 2)).tolist(),
                        radius_mean=float(radii.mean()), radius_max=float(radii.max()),
                        radius_cv=float(radii.std(unbiased=False) / radii.mean().clamp_min(1e-6)),
                        query_clip_fraction=float((radii >= RADIUS - 1e-5).float().mean()),
                        pair_clip_fraction=float(clip_fraction[index]),
                        signed_difference_rms=float(signed_diff.square().mean().sqrt()),
                        unsigned_difference_rms=float(unsigned.square().mean().sqrt()),
                        compressed_difference_rms=float(prototypes[index][1].square().mean().sqrt()),
                        bank_attention_mass=probability.reshape(probability.shape[0], self.heads,
                            height * width, -1, 2).sum(-2).mean((0, 2)).tolist())
                    if index < 2:
                        diag.update(boundary_mean=float(boundary.sigmoid().mean()),
                                    detail_highpass_rms=float(high.square().mean().sqrt()))
                    diagnostics[f"stage{index + 1}"] = diag
        if self.collect_diag:
            self.last_diag = diagnostics
        return enhanced, dict(guide_logits=tuple(guide), boundary_logits=tuple(boundaries))

"""Curvature-one Poincare operations with tangent clipping at radius 2.3."""
import math

import torch

RADIUS = math.tanh(2.3)


def project(x):
    norm = torch.linalg.vector_norm(x, dim=-1, keepdim=True).clamp_min(1e-8)
    return x * (RADIUS / norm).clamp_max(1)


def expmap0(v):
    norm = torch.linalg.vector_norm(v, dim=-1, keepdim=True).clamp_min(1e-8)
    return v * (norm.clamp_max(2.3).tanh() / norm)


def logmap0(x):
    norm = torch.linalg.vector_norm(x, dim=-1, keepdim=True).clamp(1e-8, RADIUS)
    return x * (torch.atanh(norm) / norm)


def displacement(x, y):
    """Clipped Mobius displacement (-y) (+) x, exactly zero for x == y."""
    delta = x - y
    delta2 = delta.square().sum(-1, keepdim=True)
    y2 = y.square().sum(-1, keepdim=True)
    yd = (y * delta).sum(-1, keepdim=True)
    denominator = (1 - y2).square() - 2 * (1 - y2) * yd + y2 * delta2
    return project(((1 - y2) * delta - delta2 * y) / denominator.clamp_min(1e-8))


def aligned_distance(x, y):
    numerator = torch.linalg.vector_norm(x - y, dim=-1)
    denominator = ((1 - x.square().sum(-1)) * (1 - y.square().sum(-1))).clamp_min(1e-8).sqrt()
    return 2 * torch.asinh(numerator / denominator)


def pairwise_distance(x, y):
    x2, y2 = x.square().sum(-1), y.square().sum(-1)
    squared = (x2.unsqueeze(-1) + y2.unsqueeze(-2) - 2 * (x @ y.transpose(-1, -2))).clamp_min(0)
    denominator = ((1 - x2).unsqueeze(-1) * (1 - y2).unsqueeze(-2)).clamp_min(1e-8)
    return 2 * (torch.asinh((squared / denominator + 1e-8).sqrt()) - math.asinh(1e-4))


def to_lorentz(x):
    squared = x.square().sum(-1, keepdim=True)
    denominator = (1 - squared).clamp_min(1e-8)
    return torch.cat(((1 + squared) / denominator, 2 * x / denominator), dim=-1)


def from_lorentz_sum(x):
    """Normalize a positive Lorentz sum and return its Poincare centroid."""
    time, space = x[..., :1], x[..., 1:]
    norm = (time.square() - space.square().sum(-1, keepdim=True)).clamp_min(1e-8).sqrt()
    return project(space / (time + norm).clamp_min(1e-8))


def weighted_centroid(weights, values):
    """Einstein/Lorentz centroid of manifold values, not a Euclidean mean."""
    return from_lorentz_sum(weights @ to_lorentz(values))

"""Tensor operations and auxiliary supervision for paired pyramid contexts."""
import math

import torch
from torch import nn
from torch.nn import functional as F


def channel_norm(layer, x):
    return layer(x.permute(0, 2, 3, 1)).permute(0, 3, 1, 2)


def regions(x, grid):
    """B,H,height,width,D -> B,H,regions,pixels,D."""
    batch, heads, height, width, dim = x.shape
    return x.reshape(batch, heads, grid, height // grid, grid, width // grid, dim).permute(
        0, 1, 2, 4, 3, 5, 6).reshape(batch, heads, grid * grid, -1, dim)


def feature_regions(x, grid):
    return regions(x.permute(0, 2, 3, 1).unsqueeze(1), grid).squeeze(1)


def split_heads(x, heads):
    batch, channels, height, width = x.shape
    return x.reshape(batch, heads, channels // heads, height, width).permute(0, 1, 3, 4, 2)


def merge_heads(x, height, width):
    return x.transpose(-1, -2).reshape(x.shape[0], -1, height, width)


def project_tokens(layer, tokens, heads):
    projected = F.linear(tokens, layer.weight[:, :, 0, 0])
    return projected.reshape(projected.shape[0], -1, heads, projected.shape[-1] // heads).transpose(1, 2)


def source_bias(bias, query_scale, counts):
    """Give each source scale equal prior mass before its learned bias."""
    return torch.cat([(bias[:, query_scale, scale] - math.log(count))[:, None].expand(-1, count)
                      for scale, count in enumerate(counts)], dim=-1)[None, :, None]


def binomial_highpass(x):
    kernel = x.new_tensor([[1, 2, 1], [2, 4, 2], [1, 2, 1]]) / 16
    blurred = F.conv2d(F.pad(x, (1, 1, 1, 1), mode="reflect"),
                       kernel.expand(x.shape[1], 1, 3, 3), groups=x.shape[1])
    return x - blurred


def attention_diagnostics(probability, counts):
    tv = (probability - probability.mean(-2, keepdim=True)).abs().sum(-1) / 2
    return dict(
        entropy=float(-(probability * probability.clamp_min(1e-9).log()).sum(-1).mean()
                      / math.log(probability.shape[-1])),
        scale_mass=torch.stack([p.sum(-1) for p in probability.split(counts, -1)], -1).mean((0, 2)).tolist(),
        query_attention_tv=float(tv.mean()),
        query_attention_tv_per_head=tv.mean((0, 2)).tolist())


def residual_diagnostics(feature, context, local, gamma_context, gamma_detail):
    def rms(x):
        return x.square().mean().sqrt()
    base = rms(feature).clamp_min(1e-8)
    context_residual = gamma_context * context
    local_residual = gamma_detail * local if local is not None else torch.zeros_like(feature)
    return dict(
        gamma_context=float(gamma_context), gamma_detail=float(gamma_detail),
        branch_rms=float(rms(context)), context_rms=float(rms(context)),
        local_rms=float(rms(local)) if local is not None else 0.,
        feature_rms=float(base), residual_rms=float(rms(context_residual + local_residual)),
        residual_to_feature_rms=float(rms(context_residual + local_residual) / base),
        context_residual_to_feature_rms=float(rms(context_residual) / base),
        local_residual_to_feature_rms=float(rms(local_residual) / base),
        finite=bool(torch.isfinite(context).all() and
                    (local is None or torch.isfinite(local).all())))


class AuxiliaryContext(nn.Module):
    """CTRL retains guide and boundary supervision on the TAR representation."""

    geometry = "none"
    value_geometry = "euclidean"
    grid_sizes = (8, 4, 4, 4)
    tokens_per_scale = (0, 0, 0, 0)

    def __init__(self, channels=64):
        super().__init__()
        self.norm = nn.LayerNorm(channels)
        self.guide = nn.Conv2d(channels, 1, 1)
        self.boundary = nn.Conv2d(channels, 1, 1)
        self.collect_diag = False
        self.last_diag = {}

    def forward(self, features_a, features_b, tar):
        normalized = [channel_norm(self.norm, x) for x in tar]
        return list(tar), dict(guide_logits=tuple(self.guide(x) for x in normalized),
                               boundary_logits=tuple(self.boundary(x) for x in normalized[:2]))
